# ============================================================================
# AI Agent Scheduler - Crontab Configuration
# ============================================================================
#
# SETUP STEPS:
#   1. Make sure deploy.sh has been run at least once so flaskapp-repo/ exists
#   2. Test manually first:
#      cd /home/fieldsprout/flaskapp-repo
#      /home/fieldsprout/virtualenv/flaskapp/3.9/bin/python3 run_agents.py --layer tactical
#   3. Add to crontab: crontab -e, then paste the lines below
#
# Schedule:
#   - Tactical agents: Every hour (keyword optimization, negative keywords)
#   - Operational agents: Every 4 hours (campaign monitoring, budget guardian)
#   - Strategic agent: Daily at 6am (budget reallocation, scaling decisions)
# ============================================================================

PYTHON=/home/fieldsprout/virtualenv/flaskapp/3.9/bin/python3
AGENTS=/home/fieldsprout/run_agents.py
LOGS=/home/fieldsprout/logs/agents

# ============================================================================
# TACTICAL AGENTS - Run Every Hour
# ============================================================================
# These agents make frequent small optimizations:
#   - Keyword Optimizer: Pauses expensive non-converting keywords
#   - Negative Keyword Agent: Blocks "free", "DIY", "jobs" searches
#   - Ad Copy Scientist: Tests ad variations

0 * * * * $PYTHON $AGENTS --layer tactical >> $LOGS-tactical.log 2>&1

# ============================================================================
# OPERATIONAL AGENTS - Run Every 4 Hours
# ============================================================================
# These agents monitor for anomalies and protect your budget:
#   - Campaign Manager: Detects CPL spikes and conversion drops
#   - Budget Guardian: Emergency pauses for runaway spend
#   - Quality Score Doctor: Diagnoses and fixes low Quality Scores

0 */4 * * * $PYTHON $AGENTS --layer operational >> $LOGS-operational.log 2>&1

# ============================================================================
# STRATEGIC AGENT - Run Daily at 6am
# ============================================================================
# This agent makes high-level budget and scaling decisions:
#   - Strategic Director: Budget reallocation, campaign scaling, pause decisions

0 6 * * * $PYTHON $AGENTS --layer strategic >> $LOGS-strategic.log 2>&1

# ============================================================================
# NEGATIVE KEYWORD AGENT - Run Daily at 3am
# ============================================================================
# Dedicated negative keyword sweep across all campaigns

0 3 * * * $PYTHON $AGENTS --layer negative_keyword >> $LOGS-negative_keyword.log 2>&1
